\begin{algorithm}[H]
\caption{Multi-Checkpoint Fingerprint Collection}
\begin{algorithmic}[1]
\REQUIRE context $c$, token $t$, model $\theta$, checkpoints $\mathcal{C}=\{c_1<\dots<c_m=L\}$, projection dim $k\ll D$, projection seed $\sigma'$
\STATE $a^{(1)},\dots,a^{(L)} \leftarrow \mathrm{ForwardPass}(\theta,c,t)$
\FOR{$c_i \in \mathcal{C}$}
  \STATE $f_i \leftarrow P_{\sigma'+c_i}\, a^{(c_i)}$
\ENDFOR
\RETURN $f = (f_1,\dots,f_m) \in \mathbb{R}^{mk}$
\end{algorithmic}
\end{algorithm}
\end{minipage}\hfill
\begin{minipage}[t]{0.48\textwidth}
\vspace{0pt}
\begin{algorithm}[H]
\caption{Randomized-Depth Fingerprint Matching}
\begin{algorithmic}[1]
\REQUIRE context $c$, token $t$, model $\theta$, checkpoints $\mathcal{C}$, projection dim $k$, seed $\sigma'$, audit distribution $\pi$ over $\mathcal{C}$, reference fingerprints $f\in\mathbb{R}^{mk}$
\STATE $d \sim \pi$
\STATE $\hat a^{(1)},\dots,\hat a^{(d)} \leftarrow \mathrm{PartialForwardPass}(\theta,c,t,d)$
\FOR{$c_i \in \mathcal{C}$ with $c_i \le d$}
  \STATE $\hat f_i \leftarrow P_{\sigma'+c_i}\, \hat a^{(c_i)}$
  \STATE $d_i \leftarrow \|f_i - \hat f_i\|_2$
\ENDFOR
\RETURN $\{d_i : c_i \le d\}$
\end{algorithmic}
\end{algorithm}
\end{minipage}
